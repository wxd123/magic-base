## 文档工具
# 文档工具安装
uv add python_docstring_markdown --dev

# 生成文档命令
python -m python_docstring_markdown ./src/magic_base ./docs/api-base.md

## 测试工具
# 安装测试工具
uv add pytest pytest-cov --dev

## 同步工具
# uv.lock同步
uv sync

# uv 导出 requirements.txt
uv pip compile pyproject.toml -o requirements.txt --no-sources


## 发布工具
# 上传pypi工具安装
uv add build twine --dev

# uv 打包，避免本地路径引用
uv build --no-sources

# 上传前检测
twine check dist/*

# 上传
twine upload dist/*

# 清理build文件
find . -type d -name "*.egg-info" -exec rm -rf {} + 2>/dev/null
